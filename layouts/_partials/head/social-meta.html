{{- /* Expects page, pageTitle, and description from head.html. Articles get their own Open Graph image; every other page shares the site default. */ -}} {{- $isArticle := and .page.IsPage (eq .page.Section "posts") -}} {{- $openGraphTitle := site.Title -}} {{- $openGraphImage := "" -}} {{- if $isArticle -}} {{- $openGraphTitle = .pageTitle -}} {{- $openGraphImage = partial "open-graph/image.html" (dict "mode" "article" "title" .pageTitle "date" (.page.Date.Format "2006-01-02") "path" (printf "open-graph/%s.png" .page.File.ContentBaseName) ) -}} {{- else -}} {{- $openGraphImage = partialCached "open-graph/image.html" (dict "mode" "default" "title" site.Title "path" "open-graph/default.png" ) "default-open-graph-image" -}} {{- end -}} {{- if $openGraphImage }} {{- end }} {{- if $isArticle }} {{- range .page.GetTerms "tags" }} {{- end }} {{- end }} {{- if $openGraphImage }} {{- end }}
