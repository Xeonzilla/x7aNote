{{- $ratioClass := "" -}} {{- $loading := "lazy" -}} {{- $fetchPriority := "" -}} {{- if eq .role "cover" -}} {{- $loading = "" -}} {{- $fetchPriority = "high" -}} {{- end -}} {{- if and .hook.IsBlock (.hook.Page.Store.Get "inImageRow") -}} {{- /* image-row ratios are emitted as external CSS to satisfy strict CSP. */ -}} {{- $ratioClass = printf "image-row-ratio-%dx%d" .image.Width .image.Height -}} {{- .hook.Page.Store.SetInMap "imageRowRatios" $ratioClass (dict "class" $ratioClass "ratio" (printf "%.8f" (div (float .image.Width) (float .image.Height))) ) -}} {{- end -}} {{- return (dict "fetchPriority" $fetchPriority "loading" $loading "ratioClass" $ratioClass ) -}}
