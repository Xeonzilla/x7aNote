(in $remotePath "?") (in $remotePath "#") -}} {{- partial "fail.html" (dict "area" "remote image" "problem" "unsupported remote image path" "location" .Page.File.Path "expected" "a non-empty path without trailing slash, .., query, or fragment" "got" (printf "%q" $remotePath) "fix" "use a direct image path under params.remote_images.base_url" ) -}} {{- else -}} {{- $destinationHash := $destination | crypto.Hash "sha256" -}} {{- $resolved = dict "destination" $destination "hash" $destinationHash "location" .Page.File.Path "options" (dict "headers" (dict $siteConfig.remoteImages.fetchHeaderName $siteConfig.remoteImages.fetchHeaderValue) "key" (printf "remote-images/%s" $destinationHash) ) -}} {{- end -}} {{- end -}} {{- return $resolved -}}
