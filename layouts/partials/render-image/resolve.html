{{- $sourcePath := .Destination -}} {{- $siteConfig := partialCached "site-config.html" site "site-config" -}} {{- $baseURL := $siteConfig.remoteImages.baseURL -}} {{- $resolved := "" -}} {{- $sourceURL := urls.Parse $sourcePath -}} {{- if or $sourceURL.Scheme $sourceURL.Host (strings.HasPrefix $sourcePath "/") -}} {{- partial "fail.html" (dict "area" "remote image" "problem" "non-relative image path" "location" .Page.File.Path "expected" "a relative image path without scheme, host, or leading slash" "got" (printf "%q" $sourcePath) "fix" "replace the URL with a relative image path" ) -}} {{- else -}} {{- $remotePath := printf "%s/%s" .Page.File.BaseFileName $sourcePath -}} {{- if or (not $remotePath) (strings.HasSuffix $remotePath "/") (in (split $remotePath "/") "..") (in $remotePath "?") (in $remotePath "#") -}} {{- partial "fail.html" (dict "area" "remote image" "problem" "unsupported remote image path" "location" .Page.File.Path "expected" "a non-empty path without trailing slash, .., query, or fragment" "got" (printf "%q" $remotePath) "fix" "use a direct relative image path without traversal, query, or fragment" ) -}} {{- else -}} {{- $destination := printf "%s/%s" $baseURL $remotePath -}} {{- $destinationHash := $destination | crypto.Hash "sha256" -}} {{- $resolved = dict "destination" $destination "hash" $destinationHash "location" .Page.File.Path "options" (dict "headers" (dict $siteConfig.remoteImages.fetchHeaderName $siteConfig.remoteImages.fetchHeaderValue) "key" (printf "remote-images/%s" $destinationHash) ) -}} {{- end -}} {{- end -}} {{- return $resolved -}}
